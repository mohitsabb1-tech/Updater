  var reportsData = [];

  // Progress state
  var currentJobId      = null;
  var progressPollTimer = null;
  var elapsedTimer      = null;
  var uploadStartTime   = 0;
  var lastServerPct     = 0;
  var clientPct         = 0;
  var lastUpdateTime    = 0;
  var stuckShown        = false;

  // ─── INIT ───────────────────────────────────────────
  window.onload = function() {
    google.script.run
      .withSuccessHandler(function(res) {
        var sel = document.getElementById('reportSelect');
        if (res.error) { sel.innerHTML = '<option value="">❌ ' + res.error + '</option>'; return; }
        reportsData = res.reports;
        sel.innerHTML = '<option value="">— Select report —</option>';
        res.reports.forEach(function(r, i) {
          var opt = document.createElement('option');
          opt.value = i; opt.textContent = r.name;
          sel.appendChild(opt);
        });
        sel.disabled = false;
      })
      .withFailureHandler(function() {
        document.getElementById('reportSelect').innerHTML = '<option value="">❌ Settings load nahi hui</option>';
      })
      .getReportSettings();
  };

  function onReportChange() {
    var idx = document.getElementById('reportSelect').value;
    var top = document.getElementById('topBadge');
    if (idx === '') { top.style.display = 'none'; checkReady(); return; }
    var report = reportsData[parseInt(idx)];
    var type   = report.type.toUpperCase();
    top.style.display = 'inline-block';
    top.textContent   = type;
    top.className     = 'topbar-badge ' + (type === 'COMPACT' ? 'badge-compact' : 'badge-detailed');
    applySavedVoucherTypes(report);
    checkReady();
  }

  function applySavedVoucherTypes(report) {
    var tags = document.querySelectorAll('.vtag');
    if (!tags || tags.length === 0) return;
    tags.forEach(function(t){ t.classList.remove('selected'); });
    if (report.savedVoucherTypes && report.savedVoucherTypes.length > 0) {
      var saved = {};
      report.savedVoucherTypes.forEach(function(v){ saved[v.trim().toLowerCase()] = 1; });
      tags.forEach(function(tag){
        if (saved[tag.dataset.type.trim().toLowerCase()]) tag.classList.add('selected');
      });
    }
  }

  function openPicker(e) {
    if (e) e.stopPropagation();
    document.getElementById('fileInput').click();
  }

  function handleFile(input) {
    if (!input.files || !input.files.length) return;
    readFile(input.files[0]);
  }

  function readFile(file) {
    var r = new FileReader();
    r.onload = function(e) {
      var text = e.target.result;
      if (text.charCodeAt(0) === 65279 || text.indexOf('\u0000') !== -1) {
        var r2 = new FileReader();
        r2.onload = function(e2) { processFile(e2.target.result, file.name); };
        r2.readAsText(file, 'UTF-16');
      } else { processFile(text, file.name); }
    };
    r.readAsText(file, 'UTF-8');
  }

  function processFile(text, filename) {
    loadedData = text;
    var parsed = tryParse(text);
    var valid  = getValidRows(parsed);
    var dz = document.getElementById('dropZone');
    dz.classList.add('loaded');
    var sizeKB = Math.round(text.length / 1024);
    document.getElementById('loadedName').textContent = '📄 ' + filename + ' (' + valid.length + ' rows, ' + sizeKB + ' KB)';
    document.getElementById('changeLink').classList.add('show');
    renderPreview(valid);
    checkReady();
    hideProgress();
  }

  var dz = document.getElementById('dropZone');
  dz.addEventListener('dragover',  function(e){ e.preventDefault(); e.stopPropagation(); dz.classList.add('dragover'); });
  dz.addEventListener('dragleave', function(e){ if (!dz.contains(e.relatedTarget)) dz.classList.remove('dragover'); });
  dz.addEventListener('drop', function(e) {
    e.preventDefault(); e.stopPropagation();
    dz.classList.remove('dragover');
    var file = e.dataTransfer.files[0];
    if (file) readFile(file);
  });

  function tryParse(raw) {
    try {
      var f = raw.replace(/[\u0000-\u0008\u000B\u000C\u000E-\u001F]/g, '');
      f = f.replace(/\]\s*,\s*"ItemRow"\s*:\s*\[/g, ',');
      return JSON.parse(f);
    } catch(e) { return null; }
  }

  function getValidRows(parsed) {
    if (!parsed) return [];
    var env = parsed.ENVELOPE || parsed.Envelope || parsed;
    var items = env.ItemRow || env.ITEMROW || [];
    if (!Array.isArray(items)) items = [items];
    return items.filter(function(r){ return r && (r.BSODateData || r.BSOItemData); });
  }

  // ─── PREVIEW with smart duplicate detection ──────────
  function renderPreview(valid) {
    var body = document.getElementById('previewBody');
    if (!valid.length) {
      body.innerHTML = '<div class="preview-empty"><div class="preview-empty-icon">⚠️</div><div class="preview-empty-text">Koi valid data nahi mila</div></div>';
      return;
    }

    function pd(s) {
      if (!s) return null;
      var p = s.split('-');
      return p.length === 3 ? new Date(+p[2], +p[1]-1, +p[0]) : null;
    }
    var MS = ['Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec'];
    function fmt(d) { if (!d) return '—'; var dd = d.getDate(); var mm = d.getMonth(); var yy = d.getFullYear(); return (dd < 10 ? '0'+dd : dd) + '-' + MS[mm] + '-' + yy; }

    var dates = valid.map(function(r){ return pd(r.BSODateData); }).filter(Boolean);
    dates.sort(function(a,b){ return a-b; });

    // SMART unique bills: bill+party+yymm = unique key
    var billKeys = {};
    valid.forEach(function(r){
      var bill = (r.BSOBillNoData || '').toString().trim();
      var party = (r.BSOPartyData || '').toString().trim();
      if (!bill || !party) return;
      var d = pd(r.BSODateData);
      var ym = d ? (d.getFullYear() * 100 + d.getMonth()) : '';
      var key = bill.toLowerCase() + '||' + party.toLowerCase() + '||' + ym;
      if (!billKeys[key]) billKeys[key] = { bill: bill, party: party, date: r.BSODateData };
    });
    var uniqueBillCount = Object.keys(billKeys).length;

    var types = {};
    valid.forEach(function(r){ if(r.BSOVoucherTypeData) types[r.BSOVoucherTypeData] = 1; });
    var typeKeys = Object.keys(types);

    var totalVal = 0;
    valid.forEach(function(r){ totalVal += Math.abs(parseFloat(r.BSOTotalValueData) || 0); });

    body.innerHTML =
      '<div class="stat-grid">' +
        '<div class="stat-cell"><div class="stat-label">Total Rows</div><div class="stat-val green">' + valid.length.toLocaleString() + '</div></div>' +
        '<div class="stat-cell"><div class="stat-label">Unique Bills</div><div class="stat-val blue">' + uniqueBillCount + '</div></div>' +
        '<div class="stat-cell"><div class="stat-label">Total Value</div><div class="stat-val purple" style="font-size:15px">₹' + totalVal.toLocaleString('en-IN', {maximumFractionDigits:0}) + '</div></div>' +
        '<div class="stat-cell"><div class="stat-label">Voucher Types</div><div class="stat-val" style="font-size:15px">' + typeKeys.length + '</div></div>' +
      '</div>' +
      '<div class="date-range-bar">' +
        '<div class="date-from">' + fmt(dates[0]) + '</div>' +
        '<div class="date-arrow">────→</div>' +
        '<div class="date-to">' + fmt(dates[dates.length-1]) + '</div>' +
      '</div>' +
      '<div class="voucher-tags" id="voucherTags">' +
        typeKeys.map(function(t){ return '<div class="vtag" onclick="toggleVoucher(this)" data-type="' + t.replace(/"/g,"&quot;") + '">' + t + '</div>'; }).join('') +
      '</div>' +
      '<div class="vtag-actions">' +
        '<button class="vtag-action-btn" onclick="selectAllVouchers()">Select All</button>' +
        '<button class="vtag-action-btn" onclick="clearAllVouchers()">Clear All</button>' +
      '</div>' +
      '<div class="detail-list">' +
        '<div class="detail-row"><span class="detail-key">First Item</span><span class="detail-val">' + trunc(valid[0].BSOItemData, 28) + '</span></div>' +
        '<div class="detail-row"><span class="detail-key">First Party</span><span class="detail-val accent">' + trunc(valid[0].BSOPartyData, 28) + '</span></div>' +
        '<div class="detail-row"><span class="detail-key">Last Party</span><span class="detail-val accent">' + trunc(valid[valid.length-1].BSOPartyData, 28) + '</span></div>' +
      '</div>';
  }

  function trunc(s, n) { return s ? (s.length > n ? s.substring(0,n-1)+'…' : s) : '—'; }

  function toggleVoucher(el) { el.classList.toggle('selected'); autoSaveVouchers(); }
  function autoSaveVouchers() {
    var idx = document.getElementById('reportSelect').value;
    if (idx === '') return;
    var report = reportsData[parseInt(idx)];
    var selected = getSelectedVouchers();
    report.savedVoucherTypes = selected;
    google.script.run.saveVoucherTypes(report.name, selected);
  }
  function refreshPage() { location.reload(); }
  function selectAllVouchers() {
    document.querySelectorAll('.vtag').forEach(function(t){ t.classList.add('selected'); });
    autoSaveVouchers();
  }
  function clearAllVouchers() {
    document.querySelectorAll('.vtag').forEach(function(t){ t.classList.remove('selected'); });
    autoSaveVouchers();
  }
  function getSelectedVouchers() {
    var tags = document.querySelectorAll('.vtag');
    if (!tags || tags.length === 0) return [];
    var sel = [];
    tags.forEach(function(t){ if (t.classList.contains('selected')) sel.push(t.dataset.type); });
    return sel;
  }
  function checkReady() {
    document.getElementById('uploadBtn').disabled = !(loadedData.length > 0 && document.getElementById('reportSelect').value !== '');
    var idx = document.getElementById('reportSelect').value;
    if (idx !== '') applySavedVoucherTypes(reportsData[parseInt(idx)]);
  }

  // ╔══════════════════════════════════════════════════╗
  // ║  MODAL helpers (replaces native alert/confirm)   ║
  // ╚══════════════════════════════════════════════════╝
  function showModal(opts) {
    var ov  = document.getElementById('modalOverlay');
    var ic  = document.getElementById('modalIcon');
    var tt  = document.getElementById('modalTitle');
    var sb  = document.getElementById('modalSubtitle');
    var bd  = document.getElementById('modalBody');
    var ac  = document.getElementById('modalActions');
    ic.className = 'modal-icon ' + (opts.type || 'info');
    ic.textContent = opts.icon || (opts.type === 'warning' ? '⚠️' : opts.type === 'error' ? '✕' : 'ℹ');
    tt.textContent = opts.title || '';
    sb.textContent = opts.subtitle || '';
    bd.innerHTML   = opts.bodyHtml || '';
    ac.innerHTML   = '';
    (opts.buttons || [{label:'OK', style:'primary', action:hideModal}]).forEach(function(b){
      var btn = document.createElement('button');
      btn.className = 'modal-btn ' + (b.style || '');
      btn.textContent = b.label;
      btn.onclick = function(){ if (b.action) b.action(); };
      ac.appendChild(btn);
    });
    ov.classList.add('show');
  }
  function hideModal() { document.getElementById('modalOverlay').classList.remove('show'); }

  // ╔══════════════════════════════════════════════════╗
  // ║  CHUNKED UPLOAD (sheet-backed, parallel)         ║
  // ╚══════════════════════════════════════════════════╝
  // Server side me har chunk hidden temp sheet ki cell me save hota hai.
  // Sheet cell limit = 50,000 chars → hum 40,000 chars per chunk use karte hain.
  // 100 MB file = ~2,500 chunks × 40 KB. Parallel uploads = 5 at a time.
  var CHUNK_SIZE      = 40 * 1024;  // 40 KB per chunk (safe under 50K cell limit)
  var PARALLEL_CHUNKS = 5;          // 5 chunks in flight at once

  function uploadChunksParallel(jobId, totalChunks, onAllSent, onError) {
    var nextIdx = 0;
    var completed = 0;
    var failed = false;
    var inFlight = 0;

    function startNext() {
      if (failed) return;
      while (inFlight < PARALLEL_CHUNKS && nextIdx < totalChunks) {
        sendOne(nextIdx);
        nextIdx++;
      }
    }

    function sendOne(idx) {
      inFlight++;
      var start = idx * CHUNK_SIZE;
      var end   = Math.min(start + CHUNK_SIZE, loadedData.length);
      var chunk = loadedData.substring(start, end);

      // UI: dikhao kitne complete hue
      var pct = Math.round(completed / totalChunks * 8);
      setProgressBar(pct);
      var sentMB  = (completed * CHUNK_SIZE / 1024 / 1024).toFixed(1);
      var totalMB = (loadedData.length / 1024 / 1024).toFixed(1);
      setStageText(
        '📤 Uploading file (' + completed + '/' + totalChunks + ' chunks)',
        sentMB + ' MB / ' + totalMB + ' MB — Sheet me save ho raha hai'
      );
      lastUpdateTime = new Date().getTime();

      google.script.run
        .withSuccessHandler(function(r) {
          inFlight--;
          if (failed) return;
          if (!r || !r.ok) {
            failed = true;
            onError(r && r.error ? r.error : 'Chunk ' + idx + ' send fail');
            return;
          }
          completed++;
          if (completed === totalChunks) {
            setProgressBar(8);
            setStageText('✅ All chunks uploaded', 'Server processing shuru kar raha hai...');
            onAllSent();
            return;
          }
          startNext();
        })
        .withFailureHandler(function(err) {
          inFlight--;
          if (failed) return;
          var msg = (err && err.message) ? err.message : String(err);
          if (msg.indexOf('413') !== -1) {
            failed = true;
            onError('HTTP 413: Ek single chunk bhi limit cross kar gaya. Refresh karke retry karein.');
          } else if (msg.indexOf('permission') !== -1 || msg.indexOf('authorization') !== -1) {
            failed = true;
            onError('PERMISSION_NEEDED');
          } else {
            // Transient error — retry once
            setTimeout(function(){ if (!failed) sendOne(idx); }, 1500);
          }
        })
        .uploadChunk(jobId, idx, totalChunks, chunk);
    }

    startNext();
  }

  // ╔══════════════════════════════════════════════════╗
  // ║  MAIN UPLOAD FLOW                                ║
  // ╚══════════════════════════════════════════════════╝
  function startUpload(allowDuplicates) {
    if (!loadedData) {
      showModal({ type:'warning', title:'File missing', subtitle:'Pehle JSON file choose karein!',
        buttons:[{label:'OK', style:'primary', action:hideModal}] });
      return;
    }
    var idx = document.getElementById('reportSelect').value;
    if (idx === '') {
      showModal({ type:'warning', title:'Report missing', subtitle:'Report select karein!',
        buttons:[{label:'OK', style:'primary', action:hideModal}] });
      return;
    }

    var report = reportsData[parseInt(idx)];
    var btn = document.getElementById('uploadBtn');
    btn.disabled = true;
    btn.innerHTML = '<div class="spinner"></div> Uploading...';

    currentJobId = 'job_' + new Date().getTime() + '_' + Math.floor(Math.random() * 1000000);
    uploadStartTime = new Date().getTime();
    lastUpdateTime  = new Date().getTime();
    lastServerPct = 0;
    clientPct = 0;
    stuckShown = false;

    showProgress('loading', 'Preparing upload', 'File chunks me ban rahi hai...');
    document.getElementById('progressJobId').textContent = 'Job: ' + currentJobId.split('_').slice(-1)[0];

    // Elapsed timer + stuck detector
    elapsedTimer = setInterval(function(){
      var now = new Date().getTime();
      var sec = (now - uploadStartTime) / 1000;
      document.getElementById('progressElapsed').textContent = sec.toFixed(1) + 's';

      var sinceLast = (now - lastUpdateTime) / 1000;
      if (sinceLast > 8) {
        document.getElementById('stuckLastUpdate').textContent = sinceLast.toFixed(0);
        document.getElementById('progressCard').classList.add('stuck');
        stuckShown = true;
      } else if (stuckShown && sinceLast < 3) {
        document.getElementById('progressCard').classList.remove('stuck');
        stuckShown = false;
      }
    }, 200);

    // Calculate chunks
    var totalChunks = Math.ceil(loadedData.length / CHUNK_SIZE);
    var fileMB = (loadedData.length / 1024 / 1024).toFixed(1);
    setStageText(
      '📤 Uploading file (0/' + totalChunks + ' chunks)',
      fileMB + ' MB ko ' + totalChunks + ' chunks me bhej rahe hain (' + PARALLEL_CHUNKS + ' parallel, sheet-backed)'
    );

    // Step 1: send all chunks in parallel
    uploadChunksParallel(currentJobId, totalChunks,
      function onAllSent() {
        // Step 2: trigger server-side processing + start progress polling
        progressPollTimer = setInterval(pollProgress, 700);
        runProcessOnServer(report, allowDuplicates, totalChunks);
      },
      function onError(err) {
        stopProgressTimers();
        btn.disabled = false;
        btn.innerHTML = '↑ Upload to Sheet';

        if (err === 'PERMISSION_NEEDED') {
          // Special handling for permission errors
          finalizeProgress('error', '🔐 Permission required', 'Drive access ki permission grant karni hogi (ek baar)');
          showPermissionModal();
        } else {
          finalizeProgress('error', '❌ Chunk upload error', err);
        }
      }
    );
  }

  function runProcessOnServer(report, allowDuplicates, totalChunks) {
    var btn = document.getElementById('uploadBtn');
    google.script.run
      .withSuccessHandler(function(res) {
        stopProgressTimers();
        btn.disabled = false;
        btn.innerHTML = '↑ Upload to Sheet';

        if (!res) {
          finalizeProgress('error', '❌ No response', 'Server se koi response nahi mila');
          google.script.run.clearProgress(currentJobId);
          return;
        }

        // Duplicate detected — show modal
        if (res.duplicate && res.duplicates && res.duplicates.length) {
          finalizeProgress('warning', '⚠️ Duplicate bills detected', res.message || '');
          showDuplicateModal(res.duplicates, report, totalChunks);
          // Don't clear progress yet (user might re-trigger)
          return;
        }

        if (res.success) {
          finalizeProgress('success', '✅ Upload complete', res.message || '');
        } else {
          finalizeProgress('error', '❌ Upload failed', res.message || '');
        }
        google.script.run.clearProgress(currentJobId);
      })
      .withFailureHandler(function(err) {
        stopProgressTimers();
        btn.disabled = false;
        btn.innerHTML = '↑ Upload to Sheet';
        var msg = (err && err.message) ? err.message : String(err);
        if (msg.indexOf('413') !== -1) {
          msg += '\n\n💡 Solution: File abhi bhi limit cross kar rahi hai. Refresh karke chota file try karein.';
        }
        finalizeProgress('error', '❌ Server error', msg);
        google.script.run.clearProgress(currentJobId);
      })
      .processJSONChunked(currentJobId, totalChunks, report.name, report.sheet, report.sheetId, report.type.toUpperCase(), getSelectedVouchers(), !!allowDuplicates);
  }

  function showDuplicateModal(dupes, report, totalChunks) {
    var html = '<div style="font-size:11px; color:var(--muted); margin-bottom:10px;">' +
               'In bills ki same Bill No + Party + Month sheet me already mojud hai, lekin <b>alag date</b> ke saath. Yeh shayad <b>double entry</b> hai:' +
               '</div>';
    html += '<div class="dup-list">';
    dupes.forEach(function(d){
      html += '<div class="dup-item">' +
                '<div class="dup-party">🏢 ' + escapeHtml(d.party) + '</div>' +
                '<div class="dup-meta">' +
                  '<span>Bill No:</span><span><b>' + escapeHtml(d.billNo) + '</b></span>' +
                  '<span>Sheet date:</span><span>' + escapeHtml(d.existingDate) + ' (Row ' + d.existingRow + ')</span>' +
                  '<span>New date:</span><span style="color:var(--warning)"><b>' + escapeHtml(d.incomingDate) + '</b></span>' +
                '</div>' +
              '</div>';
    });
    html += '</div>';

    showModal({
      type: 'warning',
      icon: '⚠️',
      title: dupes.length + ' duplicate bill' + (dupes.length > 1 ? 's' : '') + ' mile',
      subtitle: 'Update rok diya. Aap chahein to bhi force update kar sakte hain (apne risk par).',
      bodyHtml: html,
      buttons: [
        { label: 'Cancel', style: '', action: function(){
            hideModal();
            google.script.run.clearProgress(currentJobId);
        }},
        { label: 'Force update anyway', style: 'warning', action: function(){
            hideModal();
            // Re-trigger with allowDuplicates=true (chunks already in cache, but they were cleared by assemble)
            // Need to re-upload chunks
            startUpload(true);
        }}
      ]
    });
  }

  function escapeHtml(s) {
    if (s == null) return '';
    return String(s).replace(/[&<>"']/g, function(c){
      return {'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c];
    });
  }

  // ╔══════════════════════════════════════════════════╗
  // ║  PROGRESS POLLING                                ║
  // ╚══════════════════════════════════════════════════╝
  function pollProgress() {
    if (!currentJobId) return;
    google.script.run
      .withSuccessHandler(function(p) {
        if (!p) return;
        if (typeof p.pct === 'number' && p.pct >= lastServerPct) {
          lastServerPct = p.pct;
          setProgressBar(p.pct);
        }
        if (p.stage)  setStageText(p.stage, p.detail || '');
        lastUpdateTime = new Date().getTime();
        if (stuckShown) {
          document.getElementById('progressCard').classList.remove('stuck');
          stuckShown = false;
        }
      })
      .withFailureHandler(function(){ /* ignore */ })
      .getProgress(currentJobId);
  }

  function stopProgressTimers() {
    if (progressPollTimer) { clearInterval(progressPollTimer); progressPollTimer = null; }
    if (elapsedTimer)      { clearInterval(elapsedTimer);      elapsedTimer      = null; }
  }

  function showProgress(type, stage, detail) {
    var card = document.getElementById('progressCard');
    card.className = 'progress-card show ' + type;
    setStageText(stage, detail);
    setProgressBar(0);
    document.getElementById('progressElapsed').textContent = '0.0s';
    document.getElementById('progressFinal').textContent = '';
  }

  function setStageText(stage, detail) {
    document.getElementById('progressStage').textContent  = stage  || '';
    document.getElementById('progressDetail').textContent = detail || '';
  }

  function setProgressBar(pct) {
    pct = Math.max(0, Math.min(100, pct));
    document.getElementById('progressFill').style.width = pct + '%';
    document.getElementById('progressPercent').textContent = Math.round(pct) + '%';
  }

  function finalizeProgress(type, stage, finalMessage) {
    stopProgressTimers();
    var card = document.getElementById('progressCard');
    card.className = 'progress-card show ' + type;
    document.getElementById('progressStage').textContent = stage || '';
    if (type === 'success') setProgressBar(100);
    var sec = (new Date().getTime() - uploadStartTime) / 1000;
    document.getElementById('progressElapsed').textContent = sec.toFixed(1) + 's';
    document.getElementById('progressDetail').textContent  = 'Total time: ' + sec.toFixed(1) + 's';
    document.getElementById('progressFinal').textContent   = finalMessage || '';
  }

  function hideProgress() {
    document.getElementById('progressCard').className = 'progress-card';
  }

  function showPermissionModal() {
    var html =
      '<div style="font-size:12px; line-height:1.7; color:var(--text);">' +
        '<p style="margin-bottom:12px;">Spreadsheet ka access ek baar grant karna hoga:</p>' +
        '<ol style="padding-left:20px; margin-bottom:12px; color:var(--muted);">' +
          '<li style="margin-bottom:8px;"><b style="color:var(--text)">Apps Script editor</b> kholein ' +
              '(Extensions → Apps Script)</li>' +
          '<li style="margin-bottom:8px;">Top dropdown me <b style="color:var(--accent)">setupPermissions</b> ' +
              'function select karein</li>' +
          '<li style="margin-bottom:8px;"><b style="color:var(--success)">▶ Run</b> button dabayein</li>' +
          '<li style="margin-bottom:8px;">Permission dialog → ' +
              '<b style="color:var(--success)">Allow</b> dabayein</li>' +
          '<li>Refresh karein, fir Upload retry karein</li>' +
        '</ol>' +
        '<div style="background:rgba(79,142,247,0.1); border:1px solid rgba(79,142,247,0.3); ' +
                    'border-radius:8px; padding:10px; font-size:11px; color:var(--accent);">' +

  function cancelUpload() {
    showModal({
      type: 'warning', icon: '⚠️',
      title: 'Upload cancel karein?',
      subtitle: 'Server side processing rok diya jayega aur partially uploaded chunks delete kar diye jayenge.',
      buttons: [
        { label: 'Nahi, jari rakho', style: '', action: hideModal },
        { label: 'Haan, cancel', style: 'danger', action: function(){
            hideModal();
            stopProgressTimers();
            if (currentJobId) {
              google.script.run
                .withSuccessHandler(function(){})
                .withFailureHandler(function(){})
                .cancelJob(currentJobId);
            }
            document.getElementById('uploadBtn').disabled = false;
            document.getElementById('uploadBtn').innerHTML = '↑ Upload to Sheet';
            finalizeProgress('error', '✕ Cancelled by user', 'Upload manually rok diya gaya');
        }}
      ]
    });
  }
