<script>
  var loadedData  = '';
  var reportsData = [];

  window.onload = function() {
    google.script.run
      .withSuccessHandler(function(res) {
        var sel = document.getElementById('reportSelect');
        if (res.error) { sel.innerHTML = '<option value="">❌ ' + res.error + '</option>'; return; }
        reportsData = res.reports;
        sel.innerHTML = '<option value="">— Select report —</option>';
        res.reports.forEach(function(r, i) {
          var opt = document.createElement('option');
          opt.value = i; opt.textContent = r.name;
          sel.appendChild(opt);
        });
        sel.disabled = false;
      })
      .withFailureHandler(function() {
        document.getElementById('reportSelect').innerHTML = '<option value="">❌ Settings load nahi hui</option>';
      })
      .getReportSettings();
  };

  function onReportChange() {
    var idx = document.getElementById('reportSelect').value;
    var top = document.getElementById('topBadge');

    if (idx === '') {
      top.style.display = 'none';
      checkReady(); return;
    }

    var report = reportsData[parseInt(idx)];
    var type   = report.type.toUpperCase();

    // Topbar badge update
    top.style.display = 'inline-block';
    top.textContent   = type;
    top.className     = 'topbar-badge ' + (type === 'COMPACT' ? 'badge-compact' : 'badge-detailed');

    // Voucher tags — is report ke saved types ke hisaab se re-apply karo
    applySavedVoucherTypes(report);

    checkReady();
  }

  function applySavedVoucherTypes(report) {
    var tags = document.querySelectorAll('.vtag');
    if (!tags || tags.length === 0) return; // tags abhi render nahi hue

    // Pehle sab deselect karo
    tags.forEach(function(t){ t.classList.remove('selected'); });

    // Phir saved wale select karo
    if (report.savedVoucherTypes && report.savedVoucherTypes.length > 0) {
      var saved = {};
      report.savedVoucherTypes.forEach(function(v){ saved[v.trim().toLowerCase()] = 1; });
      tags.forEach(function(tag){
        if (saved[tag.dataset.type.trim().toLowerCase()]) tag.classList.add('selected');
      });
    }
  }

  function openPicker(e) {
    if (e) e.stopPropagation();
    document.getElementById('fileInput').click();
  }

  function handleFile(input) {
    if (!input.files || !input.files.length) return;
    readFile(input.files[0]);
  }

  function readFile(file) {
    var r = new FileReader();
    r.onload = function(e) {
      var text = e.target.result;
      if (text.charCodeAt(0) === 65279 || text.indexOf('\u0000') !== -1) {
        var r2 = new FileReader();
        r2.onload = function(e2) { processFile(e2.target.result, file.name); };
        r2.readAsText(file, 'UTF-16');
      } else { processFile(text, file.name); }
    };
    r.readAsText(file, 'UTF-8');
  }

  function processFile(text, filename) {
    loadedData = text;
    var parsed = tryParse(text);
    var valid  = getValidRows(parsed);

    var dz = document.getElementById('dropZone');
    dz.classList.add('loaded');
    document.getElementById('loadedName').textContent = '📄 ' + filename + ' (' + valid.length + ' rows)';
    document.getElementById('changeLink').classList.add('show');

    renderPreview(valid);
    checkReady();
    hideStatus();
  }

  var dz = document.getElementById('dropZone');
  dz.addEventListener('dragover',  function(e){ e.preventDefault(); e.stopPropagation(); dz.classList.add('dragover'); });
  dz.addEventListener('dragleave', function(e){ if (!dz.contains(e.relatedTarget)) dz.classList.remove('dragover'); });
  dz.addEventListener('drop', function(e) {
    e.preventDefault(); e.stopPropagation();
    dz.classList.remove('dragover');
    var file = e.dataTransfer.files[0];
    if (file) readFile(file);
  });

  function tryParse(raw) {
    try {
      var f = raw.replace(/[\u0000-\u0008\u000B\u000C\u000E-\u001F]/g, '');
      f = f.replace(/\]\s*,\s*"ItemRow"\s*:\s*\[/g, ',');
      return JSON.parse(f);
    } catch(e) { return null; }
  }

  function getValidRows(parsed) {
    if (!parsed) return [];
    var env = parsed.ENVELOPE || parsed.Envelope || parsed;
    var items = env.ItemRow || env.ITEMROW || [];
    if (!Array.isArray(items)) items = [items];
    return items.filter(function(r){ return r && (r.BSODateData || r.BSOItemData); });
  }

  function renderPreview(valid) {
    var body = document.getElementById('previewBody');
    if (!valid.length) {
      body.innerHTML = '<div class="preview-empty"><div class="preview-empty-icon">⚠️</div><div class="preview-empty-text">Koi valid data nahi mila</div></div>';
      return;
    }

    function pd(s) {
      if (!s) return null;
      var p = s.split('-');
      return p.length === 3 ? new Date(+p[2], +p[1]-1, +p[0]) : null;
    }
    var MONTHS_SHORT = ['Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec'];
    function fmt(d) { if (!d) return '—'; var dd = d.getDate(); var mm = d.getMonth(); var yy = d.getFullYear(); return (dd < 10 ? '0'+dd : dd) + '-' + MONTHS_SHORT[mm] + '-' + yy; }

    var dates = valid.map(function(r){ return pd(r.BSODateData); }).filter(Boolean);
    dates.sort(function(a,b){ return a-b; });

    // Unique bills
    var bills = {};
    valid.forEach(function(r){ if(r.BSOBillNoData) bills[r.BSOBillNoData] = 1; });
    var billCount = Object.keys(bills).length;

    // Voucher types
    var types = {};
    valid.forEach(function(r){ if(r.BSOVoucherTypeData) types[r.BSOVoucherTypeData] = 1; });
    var typeKeys = Object.keys(types);

    // Total value
    var totalVal = 0;
    valid.forEach(function(r){ totalVal += Math.abs(parseFloat(r.BSOTotalValueData) || 0); });

    body.innerHTML =
      '<div class="stat-grid">' +
        '<div class="stat-cell"><div class="stat-label">Total Rows</div><div class="stat-val green">' + valid.length.toLocaleString() + '</div></div>' +
        '<div class="stat-cell"><div class="stat-label">Unique Bills</div><div class="stat-val blue">' + billCount + '</div></div>' +
        '<div class="stat-cell"><div class="stat-label">Total Value</div><div class="stat-val purple" style="font-size:15px">₹' + totalVal.toLocaleString('en-IN', {maximumFractionDigits:0}) + '</div></div>' +
        '<div class="stat-cell"><div class="stat-label">Voucher Types</div><div class="stat-val" style="font-size:15px">' + typeKeys.length + '</div></div>' +
      '</div>' +
      '<div class="date-range-bar">' +
        '<div class="date-from">' + fmt(dates[0]) + '</div>' +
        '<div class="date-arrow">────→</div>' +
        '<div class="date-to">' + fmt(dates[dates.length-1]) + '</div>' +
      '</div>' +
      '<div class="voucher-tags" id="voucherTags">' +
        typeKeys.map(function(t){ return '<div class="vtag" onclick="toggleVoucher(this)" data-type="' + t.replace(/"/g,"&quot;") + '">' + t + '</div>'; }).join('') +
      '</div>' +
      '<div class="vtag-actions">' +
        '<button class="vtag-action-btn" onclick="selectAllVouchers()">Select All</button>' +
        '<button class="vtag-action-btn" onclick="clearAllVouchers()">Clear All</button>' +
      '</div>' +
      '<div class="detail-list">' +
        '<div class="detail-row"><span class="detail-key">First Item</span><span class="detail-val">' + trunc(valid[0].BSOItemData, 28) + '</span></div>' +
        '<div class="detail-row"><span class="detail-key">First Party</span><span class="detail-val accent">' + trunc(valid[0].BSOPartyData, 28) + '</span></div>' +
        '<div class="detail-row"><span class="detail-key">Last Party</span><span class="detail-val accent">' + trunc(valid[valid.length-1].BSOPartyData, 28) + '</span></div>' +
      '</div>';
  }

  function trunc(s, n) { return s ? (s.length > n ? s.substring(0,n-1)+'…' : s) : '—'; }

  function toggleVoucher(el) {
    el.classList.toggle('selected');
    autoSaveVouchers();
  }

  function autoSaveVouchers() {
    var idx = document.getElementById('reportSelect').value;
    if (idx === '') return;
    var report = reportsData[parseInt(idx)];
    var selected = getSelectedVouchers();
    report.savedVoucherTypes = selected;
    google.script.run.saveVoucherTypes(report.name, selected);
  }

  function refreshPage() { location.reload(); }

  function selectAllVouchers() {
    document.querySelectorAll('.vtag').forEach(function(t){ t.classList.add('selected'); });
    autoSaveVouchers();
  }

  function clearAllVouchers() {
    document.querySelectorAll('.vtag').forEach(function(t){ t.classList.remove('selected'); });
    autoSaveVouchers();
  }

  function getSelectedVouchers() {
    var tags = document.querySelectorAll('.vtag');
    if (!tags || tags.length === 0) return []; // tags exist nahi karte abhi
    var sel = [];
    tags.forEach(function(t){ if (t.classList.contains('selected')) sel.push(t.dataset.type); });
    return sel;
  }



  function checkReady() {
    document.getElementById('uploadBtn').disabled = !(loadedData.length > 0 && document.getElementById('reportSelect').value !== '');

    // File load hone ke baad saved voucher types apply karo
    var idx = document.getElementById('reportSelect').value;
    if (idx !== '') {
      applySavedVoucherTypes(reportsData[parseInt(idx)]);
    }
  }

  function startUpload() {
    if (!loadedData) { showStatus('Pehle file choose karein!', 'error'); return; }
    var idx = document.getElementById('reportSelect').value;
    if (idx === '') { showStatus('Report select karein!', 'error'); return; }

    var report = reportsData[parseInt(idx)];
    var btn = document.getElementById('uploadBtn');
    btn.disabled = true;
    btn.innerHTML = '<div class="spinner"></div> Uploading...';
    showStatus('Processing...', 'loading');

    google.script.run
      .withSuccessHandler(function(res) {
        btn.disabled = false;
        btn.innerHTML = '↑ Upload to Sheet';
        showStatus(res, res && (res.includes('✅') || res.includes('Done')) ? 'success' : 'error');
      })
      .withFailureHandler(function(err) {
        btn.disabled = false;
        btn.innerHTML = '↑ Upload to Sheet';
        showStatus('Error: ' + err.message, 'error');
      })
      .processJSON(loadedData, report.name, report.sheet, report.sheetId, report.type.toUpperCase(), getSelectedVouchers());
  }

  function showStatus(msg, type) {
    var el = document.getElementById('statusMsg');
    el.textContent = msg; el.className = 'status-msg show ' + type;
  }
  function hideStatus() { document.getElementById('statusMsg').className = 'status-msg'; }
</script>
