var MONTHS_SHORT = ['Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec'];

// ╔══════════════════════════════════════════════════════════╗
// ║  ONE-TIME SETUP — run this ONCE from editor              ║

// manually permission grant karni padti hai. Yeh function editor me
// run karne par saare APIs ki permission ek saath maang lega.

function setupPermissions() {
  var msgs = [];

  // 1. Spreadsheet access test
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    msgs.push('✅ Spreadsheet access OK: ' + (ss ? ss.getName() : 'no active'));
  } catch(e) {
    msgs.push('❌ Spreadsheet: ' + e.toString());
  }

  // 2. Drive access test
  try {
    var folder = _getTempFolder();
    msgs.push('✅ Drive access OK. Temp folder: ' + folder.getName() + ' (id: ' + folder.getId() + ')');

    // Create + delete a tiny test file to confirm write permission
    var testFile = folder.createFile('permission_test_' + new Date().getTime(), 'ok', MimeType.PLAIN_TEXT);
    testFile.setTrashed(true);
    msgs.push('✅ Drive write OK (test file created & deleted)');
  } catch(e) {
    msgs.push('❌ Drive: ' + e.toString());
  }

  // 3. Cache access test
  try {
    var cache = CacheService.getUserCache();
    cache.put('setup_test', 'ok', 60);
    var v = cache.get('setup_test');
    cache.remove('setup_test');
    msgs.push('✅ Cache OK: ' + v);
  } catch(e) {
    msgs.push('❌ Cache: ' + e.toString());
  }

  // 4. Properties test
  try {
    PropertiesService.getScriptProperties();
    msgs.push('✅ Properties OK');
  } catch(e) {
    msgs.push('❌ Properties: ' + e.toString());
  }

  var report = msgs.join('\n');
  Logger.log(report);

  // Show in dialog if running from spreadsheet UI
  try {
    SpreadsheetApp.getUi().alert('Setup Status', report, SpreadsheetApp.getUi().ButtonSet.OK);
  } catch(e) {
    // Running from editor without UI - that's fine, log is enough
  }

  return report;
}

// ╔══════════════════════════════════════════════════════════╗
// ║  PROGRESS TRACKING  (CacheService)                       ║
// ╚══════════════════════════════════════════════════════════╝
function _setProgress(jobId, percent, stage, detail) {
  if (!jobId) return;
  try {
    var cache = CacheService.getUserCache();
    cache.put('prog_' + jobId, JSON.stringify({
      pct: percent,
      stage: stage || '',
      detail: detail || '',
      ts: new Date().getTime(),
      done: false
    }), 600);
  } catch(e) {}
}

function _setProgressDone(jobId, percent, stage, detail) {
  if (!jobId) return;
  try {
    CacheService.getUserCache().put('prog_' + jobId, JSON.stringify({
      pct: percent, stage: stage || '', detail: detail || '',
      ts: new Date().getTime(), done: true
    }), 600);
  } catch(e) {}
}

function getProgress(jobId) {
  if (!jobId) return null;
  try {
    var v = CacheService.getUserCache().get('prog_' + jobId);
    return v ? JSON.parse(v) : null;
  } catch(e) { return null; }
}

function clearProgress(jobId) {
  if (!jobId) return;
  try { CacheService.getUserCache().remove('prog_' + jobId); } catch(e) {}
}

// ╔══════════════════════════════════════════════════════════╗
// ║  CHUNKED UPLOAD via DRIVE  (HTTP 413 fix, 100MB+ support)║
// ╚══════════════════════════════════════════════════════════╝
//
// Bade JSON files (100MB+) Apps Script ke payload limit ko cross karte hain
// → HTTP 413. CacheService bhi sirf 100KB per value hold kar sakta hai.
// Solution: client 1MB chunks bheje, server har chunk ko Drive temp folder me
// alag file ki tarah save karta hai. Last chunk ke baad sab files read karke
// concat karta hai aur process karta hai.
//
// Why Drive?
//   ✓ Per-file size unlimited (vs CacheService 100KB limit)
//   ✓ Persistent (cache expire issues nahi)
//   ✓ Fast read/write
//
var TEMP_FOLDER_NAME = 'PflanzeUploaderTemp';

function _getTempFolder() {
  var folders = DriveApp.getFoldersByName(TEMP_FOLDER_NAME);
  if (folders.hasNext()) return folders.next();
  return DriveApp.createFolder(TEMP_FOLDER_NAME);
}

function _padIdx(n) {
  return ('00000' + n).slice(-5); // padded for proper sort order
}

function _chunkFileName(jobId, idx) {
  return 'tmpchunk_' + jobId + '_' + _padIdx(idx);
}

function uploadChunk(jobId, chunkIndex, totalChunks, chunkData) {
  try {
    var folder = _getTempFolder();
    var fileName = _chunkFileName(jobId, chunkIndex);

    // Idempotent: agar yeh chunk pehle se hai (retry case) to delete karke fresh banao
    var existing = folder.getFilesByName(fileName);
    while (existing.hasNext()) existing.next().setTrashed(true);

    // Create as plain text file
    folder.createFile(fileName, chunkData, MimeType.PLAIN_TEXT);

    var pct = Math.round((chunkIndex + 1) / totalChunks * 8); // chunks 0-8% range
    var sizeKB = Math.round(chunkData.length / 1024);
    _setProgress(jobId, pct,
      '📤 Chunk ' + (chunkIndex + 1) + '/' + totalChunks,
      'Drive me save: ' + sizeKB + ' KB (total ' + Math.round((chunkIndex+1) * sizeKB / 1024) + ' MB)');
    return { ok: true, received: chunkIndex + 1, total: totalChunks };
  } catch(e) {
    return { ok: false, error: 'Chunk ' + chunkIndex + ' fail: ' + e.toString() };
  }
}

function _assembleChunksFromDrive(jobId, totalChunks) {
  var folder = _getTempFolder();
  var parts = [];
  var totalBytes = 0;

  for (var i = 0; i < totalChunks; i++) {
    if (i % 10 === 0) {
      _setProgress(jobId, 10 + Math.round(i / totalChunks * 3),
        '📦 Chunks assemble',
        'Drive se ' + (i+1) + '/' + totalChunks + ' chunks padh raha hu...');
    }
    var fileName = _chunkFileName(jobId, i);
    var files = folder.getFilesByName(fileName);
    if (!files.hasNext()) {
      _cleanupChunks(jobId, totalChunks);
      throw new Error('Chunk ' + i + ' Drive me nahi mila. Refresh karke retry karein.');
    }
    var content = files.next().getBlob().getDataAsString();
    parts.push(content);
    totalBytes += content.length;
  }

  _setProgress(jobId, 13, '📦 Chunks assemble',
    'Sab chunks jod raha hu (' + Math.round(totalBytes / 1024 / 1024) + ' MB)...');
  var assembled = parts.join('');
  parts = null; // free memory

  // Cleanup temp files (background)
  _cleanupChunks(jobId, totalChunks);

  return assembled;
}

function _cleanupChunks(jobId, totalChunks) {
  try {
    var folder = _getTempFolder();
    for (var j = 0; j < totalChunks; j++) {
      var fileName = _chunkFileName(jobId, j);
      var files = folder.getFilesByName(fileName);
      while (files.hasNext()) {
        try { files.next().setTrashed(true); } catch(e2){}
      }
    }
  } catch(e) { /* best effort */ }
}

// Manual cleanup helper — agar koi orphan chunks reh gaye purane jobs ke
function cleanupAllOrphanChunks() {
  try {
    var folder = _getTempFolder();
    var files = folder.getFilesByName ? folder.getFiles() : null;
    if (!files) return 'No folder';
    var count = 0;
    var now = new Date().getTime();
    while (files.hasNext()) {
      var f = files.next();
      var name = f.getName();
      if (name.indexOf('tmpchunk_') === 0) {
        // Delete chunks older than 1 hour
        var age = now - f.getDateCreated().getTime();
        if (age > 60 * 60 * 1000) {
          f.setTrashed(true);
          count++;
        }
      }
    }
    return 'Cleaned ' + count + ' orphan chunks';
  } catch(e) {
    return 'Error: ' + e.toString();
  }
}

// Cancel running job — delete this jobId ke saare chunks
function cancelJob(jobId) {
  if (!jobId) return { ok: false };
  try {
    var folder = _getTempFolder();
    var files = folder.getFiles();
    var count = 0;
    var prefix = 'tmpchunk_' + jobId + '_';
    while (files.hasNext()) {
      var f = files.next();
      if (f.getName().indexOf(prefix) === 0) {
        try { f.setTrashed(true); count++; } catch(e2){}
      }
    }
    clearProgress(jobId);
    return { ok: true, deleted: count };
  } catch(e) {
    return { ok: false, error: e.toString() };
  }
}

function tallyDateToDisplay(s) {
  if (!s) return s;
  var str = String(s).trim();
  if (!str) return str;
  var p = str.split('-');
  if (p.length !== 3) return str;
  var d = parseInt(p[0], 10);
  var m = parseInt(p[1], 10);
  var y = parseInt(p[2], 10);
  if (isNaN(d) || isNaN(m) || isNaN(y) || m < 1 || m > 12) return str;
  return (d < 10 ? '0' + d : String(d)) + '-' + MONTHS_SHORT[m - 1] + '-' + y;
}

function dateToInt(s) {
  if (!s) return null;
  if (s instanceof Date) {
    if (isNaN(s.getTime())) return null;
    return s.getFullYear() * 10000 + (s.getMonth() + 1) * 100 + s.getDate();
  }
  var str = String(s).trim();
  if (!str) return null;
  var p = str.split('-');
  if (p.length !== 3) return null;

  var monthIdx = MONTHS_SHORT.indexOf(p[1]);
  if (monthIdx !== -1) {
    var d = parseInt(p[0], 10);
    var m = monthIdx + 1;
    var y = parseInt(p[2], 10);
    if (isNaN(d) || isNaN(y)) return null;
    return y * 10000 + m * 100 + d;
  }

  var d = parseInt(p[0], 10), m = parseInt(p[1], 10), y = parseInt(p[2], 10);
  if (isNaN(d) || isNaN(m) || isNaN(y)) return null;
  return y * 10000 + m * 100 + d;
}

function intToDisplay(n) {
  if (!n) return '?';
  var y = Math.floor(n / 10000);
  var m = Math.floor((n % 10000) / 100);
  var d = n % 100;
  return (d < 10 ? '0'+d : d) + '-' + MONTHS_SHORT[m-1] + '-' + y;
}

// Date ka year-month nikaalo (duplicate check ke liye)
function _yearMonth(s) {
  var di = dateToInt(s);
  if (!di) return null;
  return Math.floor(di / 100); // YYYYMM
}

function getReportSettings() {
  try {
    var ss       = SpreadsheetApp.getActiveSpreadsheet();
    var settings = ss.getSheetByName('App_Settings');
    if (!settings) return { error: "App_Settings sheet nahi mili!" };

    var lastRow = settings.getLastRow();
    if (lastRow < 2) return { error: "App_Settings mein koi data nahi." };

    var range = settings.getRange(2, 8, lastRow - 1, 5).getValues();
    var reports = [];

    range.forEach(function(row) {
      var name       = String(row[0] || "").trim();
      var sheet      = String(row[1] || "").trim();
      var sheetId    = String(row[2] || "").trim();
      var type       = String(row[3] || "").trim();
      var savedTypes = String(row[4] || "").trim();

      var voucherTypes = savedTypes
        ? savedTypes.split(",").map(function(v){ return v.trim(); }).filter(Boolean)
        : [];

      if (name) {
        reports.push({ name: name, sheet: sheet, sheetId: sheetId, type: type, savedVoucherTypes: voucherTypes });
      }
    });

    return { reports: reports };
  } catch(e) {
    return { error: e.toString() };
  }
}

function saveVoucherTypes(reportName, voucherTypes) {
  try {
    var ss       = SpreadsheetApp.getActiveSpreadsheet();
    var settings = ss.getSheetByName('App_Settings');
    if (!settings) return "Error: App_Settings nahi mili";

    var lastRow = settings.getLastRow();
    var range   = settings.getRange(2, 8, lastRow - 1, 1).getValues();

    for (var i = 0; i < range.length; i++) {
      var name = String(range[i][0] || "").trim();
      if (name === reportName) {
        settings.getRange(i + 2, 12).setValue(voucherTypes.join(", "));
        return "Saved";
      }
    }
    return "Error: Report nahi mila";
  } catch(e) {
    return "Error: " + e.toString();
  }
}

// ║  Bill grouping (COMPACT)                                  ║
// ╚══════════════════════════════════════════════════════════╝
function makeBillKey(item) {
  return [
    item.BSODateData        || "",
    item.BSOBillNoData      || "",
    item.BSOPartyData       || "",
    item.BSOVoucherTypeData || ""
  ].join("||");
}

function groupByBill(itemRows, jobId) {
  var order = [], map = {};
  var total = itemRows.length;
  itemRows.forEach(function(item, idx) {
    if (idx % 200 === 0) {
      _setProgress(jobId, 50 + Math.round(idx / total * 8),
        '🧾 Bills group ho rahe',
        'Row ' + (idx+1) + '/' + total + ' — ' + (item.BSOPartyData || '?'));
    }
    if (!item || (!item.BSODateData && !item.BSOItemData)) return;
    var key = makeBillKey(item);
    if (!map[key]) {
      order.push(key);
      map[key] = {
        date:    item.BSODateData        || "",
        billNo:  item.BSOBillNoData      || "",
        party:   item.BSOPartyData       || "",
        vchType: item.BSOVoucherTypeData || "",
        narr:    item.BSONarrationData   || "",
        items:   []
      };
    }
    if (!map[key].narr && item.BSONarrationData) map[key].narr = item.BSONarrationData;
    if (item.BSOItemData) {
      map[key].items.push({
        Itm:  item.BSOItemData       || "",
        Bch:  item.BSOBatchNoData    || "",
        Mfg:  tallyDateToDisplay(item.BSOMfgDateData  || ""),
        Exp:  tallyDateToDisplay(item.BSOExpDateData  || ""),
        MRP:  item.BSOMRPData        || "",
        QIn:  item.BSOQtyInData      || 0,
        QOut: item.BSOQtyOutData     || 0,
        Rt:   item.BSORateData       || 0,
        Val:  item.BSOTotalValueData || 0,
        Tax:  item.BSOTaxAmountData  || 0
      });
    }
  });
  return order.map(function(k) { return map[k]; });
}

// ╔══════════════════════════════════════════════════════════╗
// ║  DUPLICATE DETECTION                                      ║
// ║  Same (BillNo + Party + YearMonth) = duplicate            ║
// ╚══════════════════════════════════════════════════════════╝
function _detectDuplicatesInUpload(items, jobId) {
  // Step 1: incoming items me hi check karo
  _setProgress(jobId, 42, '🔍 Duplicate check', 'Naye data me duplicate dhund raha hu...');
  var seen = {}; // key → first occurrence (party+billNo+yymm+row)
  var dupes = [];
  items.forEach(function(item, idx) {
    var bill = String(item.BSOBillNoData || "").trim();
    var party = String(item.BSOPartyData || "").trim();
    var ym = _yearMonth(item.BSODateData);
    if (!bill || !party || !ym) return;
    var key = bill.toLowerCase() + '||' + party.toLowerCase() + '||' + ym;
    if (seen[key]) {
      // Same key, agar item alag hai (multi-item bill) to skip; check first row diff
      // Hum sirf TRUE duplicate maante hain jab dono entries me different row hi same bill+party+month repeat ho
      // Actually: ek bill me multiple items ho sakte hain, woh duplicate nahi.
      // Real duplicate: same key ke liye ALAG ALAG date wala entry, ya same date par alag total
      var first = seen[key];
      if (first.date !== item.BSODateData ||
          String(first.total) !== String(item.BSOTotalValueData || "") ||
          String(first.itemKey) !== String(item.BSOItemData || "") + '|' + (item.BSOBatchNoData || "")) {
        // Same bill+party+month but different date OR different item-batch combo with same total mismatch
        // Yeh humare liye warning case nahi hai (ek bill me multi-item normal hai).
        // Real duplicate detection: same bill+party+month + same item+batch → asli double entry
      }
    } else {
      seen[key] = {
        date: item.BSODateData,
        total: item.BSOTotalValueData,
        itemKey: (item.BSOItemData || "") + '|' + (item.BSOBatchNoData || "")
      };
    }
  });
  return dupes;
}

// Detect duplicates AGAINST EXISTING SHEET DATA
// Compares incoming bill keys against rows already in sheet.
// If incoming (bill+party+yymm) already exists in sheet AND incoming date != sheet date,
// → block + report.
function _detectDuplicatesAgainstSheet(sheet, incomingItems, type, jobId) {
  _setProgress(jobId, 46, '🔍 Sheet me duplicate check', 'Purane data se compare ho raha hai...');
  var existing = sheet.getDataRange().getValues();
  if (existing.length <= 1) return [];

  // Sheet me column mapping (COMPACT vs DETAILED)
  // Both me: Date col=0, BillNo col=1
  // COMPACT: Party col=2 ; DETAILED: Party col=13
  var partyCol = (type === 'COMPACT') ? 2 : 13;

  // Existing keys: bill + party + yymm → date
  var existingMap = {};
  for (var i = 1; i < existing.length; i++) {
    var bill = String(existing[i][1] || "").trim();
    var party = String(existing[i][partyCol] || "").trim();
    var ym = _yearMonth(existing[i][0]);
    if (!bill || !party || !ym) continue;
    var k = bill.toLowerCase() + '||' + party.toLowerCase() + '||' + ym;
    if (!existingMap[k]) {
      existingMap[k] = {
        date: existing[i][0],
        party: party,
        bill: bill,
        rowNum: i + 1
      };
    }
  }

  // Build incoming unique bill keys (one entry per bill, not per item)
  var incomingBills = {};
  incomingItems.forEach(function(item) {
    var bill = String(item.BSOBillNoData || "").trim();
    var party = String(item.BSOPartyData || "").trim();
    var ym = _yearMonth(item.BSODateData);
    if (!bill || !party || !ym) return;
    var k = bill.toLowerCase() + '||' + party.toLowerCase() + '||' + ym;
    if (!incomingBills[k]) {
      incomingBills[k] = {
        date: item.BSODateData,
        party: party,
        bill: bill
      };
    }
  });

  // Compare: a duplicate is when SAME key exists in BOTH but with DIFFERENT date
  // (same date = re-upload of same bill, that's allowed because writeToSheet replaces date range)
  var dupes = [];
  Object.keys(incomingBills).forEach(function(k) {
    if (existingMap[k]) {
      var inc = incomingBills[k];
      var exi = existingMap[k];
      var incDi = dateToInt(inc.date);
      var exiDi = dateToInt(exi.date);
      if (incDi && exiDi && incDi !== exiDi) {
        dupes.push({
          billNo: inc.bill,
          party: inc.party,
          incomingDate: tallyDateToDisplay(inc.date),
          existingDate: (exi.date instanceof Date)
            ? tallyDateToDisplay(Utilities.formatDate(exi.date, Session.getScriptTimeZone(), 'd-M-yyyy'))
            : String(exi.date),
          existingRow: exi.rowNum
        });
      }
    }
  });
  return dupes;
}

// ╔══════════════════════════════════════════════════════════╗
// ║  Sheet writer (with progress)                             ║
// ╚══════════════════════════════════════════════════════════╝
function writeToSheet(sheet, headers, newRows, rangeStart, rangeEnd, jobId) {
  _setProgress(jobId, 65, '📖 Existing sheet read',
    'Sheet ka purana data padha ja raha hai...');
  var existing = sheet.getDataRange().getValues();

  if (existing.length <= 1) {
    _setProgress(jobId, 85, '🧹 Sheet clear',
      'Khali sheet, directly likh rahe hain...');
    sheet.clearContents();
    _setProgress(jobId, 92, '✏️ Writing rows',
      newRows.length + ' rows likhi ja rahi hain...');
    var out = [headers].concat(newRows);
    sheet.getRange(1, 1, out.length, headers.length).setValues(out);
    SpreadsheetApp.flush();
    return;
  }

  _setProgress(jobId, 72, '🔍 Filtering old rows',
    (existing.length - 1) + ' purane rows me se range wale hata raha hu...');
  var kept = [];
  for (var i = 1; i < existing.length; i++) {
    var di = dateToInt(existing[i][0]);
    if (di === null || di < rangeStart || di > rangeEnd) kept.push(existing[i]);
  }

  _setProgress(jobId, 80, '📊 Sorting',
    (kept.length + newRows.length) + ' rows date-wise sort ho rahi hain...');
  var combined = kept.concat(newRows);
  combined.sort(function(a, b) {
    var da = dateToInt(a[0]), db = dateToInt(b[0]);
    if (da === null && db === null) return 0;
    if (da === null) return 1;
    if (db === null) return -1;
    return da - db;
  });

  // Column count normalize karo (kept rows me extra/kam cols ho sakte hain)
  var colCount = headers.length;
  for (var k = 0; k < combined.length; k++) {
    var row = combined[k];
    if (row.length < colCount) {
      while (row.length < colCount) row.push('');
    } else if (row.length > colCount) {
      combined[k] = row.slice(0, colCount);
    }
  }

  _setProgress(jobId, 88, '🧹 Sheet clear',
    'Purana data clear ho raha hai...');
  sheet.clearContents();

  _setProgress(jobId, 94, '✏️ Writing rows',
    'Sheet me ' + combined.length + ' rows likhi ja rahi hain...');
  var final = [headers].concat(combined);
  sheet.getRange(1, 1, final.length, colCount).setValues(final);
  SpreadsheetApp.flush();
}

// ╔══════════════════════════════════════════════════════════╗
// ║  MAIN: processJSONChunked  (called after all chunks sent) ║
// ╚══════════════════════════════════════════════════════════╝
function processJSONChunked(jobId, totalChunks, reportName, sheetName, sheetId, type, selectedVoucherTypes, allowDuplicates) {
  var lastStage = 'Initial';
  try {
    _setProgress(jobId, 10, '📦 Chunks assemble',
      totalChunks + ' chunks Drive se padhe ja rahe hain...');
    lastStage = 'Chunk assembly from Drive';
    var jsonData = _assembleChunksFromDrive(jobId, totalChunks);

    return _processJSONInternal(jsonData, reportName, sheetName, sheetId, type, selectedVoucherTypes, allowDuplicates, jobId);
  } catch(e) {
    _setProgressDone(jobId, 100, '❌ Error in: ' + lastStage, e.toString());
    return { success: false, message: '❌ ' + lastStage + ' me error: ' + e.toString() };
  }
}

// Backwards-compat: old single-shot upload
function processJSON(jsonData, reportName, sheetName, sheetId, type, selectedVoucherTypes, jobId, allowDuplicates) {
  return _processJSONInternal(jsonData, reportName, sheetName, sheetId, type, selectedVoucherTypes, allowDuplicates, jobId);
}

function _processJSONInternal(jsonData, reportName, sheetName, sheetId, type, selectedVoucherTypes, allowDuplicates, jobId) {
  var stage = 'Start';
  var lastItem = '';
  try {
    stage = 'JSON cleaning';
    _setProgress(jobId, 14, '🧼 JSON clean', 'Invalid characters hata raha hu...');
    jsonData = jsonData.replace(/[\u0000-\u0008\u000B\u000C\u000E-\u001F]/g, '');
    jsonData = jsonData.replace(/\]\s*,\s*"ItemRow"\s*:\s*\[/g, ',');

    stage = 'JSON parse';
    _setProgress(jobId, 18, '📑 JSON parse',
      'Size: ' + Math.round(jsonData.length / 1024) + ' KB');
    var data;
    try { data = JSON.parse(jsonData); }
    catch(parseErr) {
      _setProgressDone(jobId, 100, '❌ JSON parse fail', parseErr.toString());
      return { success: false, message: "❌ JSON parse error: " + parseErr.toString() };
    }

    stage = 'Sheet open';
    _setProgress(jobId, 22, '📂 Sheet open',
      'Target: ' + (sheetName || '?'));
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = null;

    if (sheetId && sheetId !== "") {
      if (sheetId.length > 15) {
        try {
          ss = SpreadsheetApp.openById(sheetId);
          sheet = ss.getSheetByName(sheetName);
        } catch(e) {
          _setProgressDone(jobId, 100, '❌ External spreadsheet fail', e.toString());
          return { success: false, message: "❌ Dusri Spreadsheet open nahi hui. File ID: " + sheetId };
        }
      } else {
        try {
          sheet = ss.getSheets().filter(function(s) {
            return s.getSheetId().toString() === sheetId.toString();
          })[0];
        } catch(e) { sheet = null; }
      }
    }

    if (!sheet && sheetName) sheet = ss.getSheetByName(sheetName);
    if (!sheet && sheetName) sheet = ss.insertSheet(sheetName);
    if (!sheet) {
      _setProgressDone(jobId, 100, '❌ Sheet nahi mili', 'Name: ' + sheetName);
      return { success: false, message: "❌ Sheet nahi ban payi! Name: '" + sheetName + "'" };
    }

    stage = 'ItemRow extract';
    _setProgress(jobId, 26, '🔎 ItemRow nikaal raha hu', '');
    var envelope = data.ENVELOPE || data.Envelope || data;
    var itemRows = envelope.ItemRow || envelope.ITEMROW;
    if (!itemRows) {
      _setProgressDone(jobId, 100, '❌ ItemRow nahi mila', 'JSON structure check karein');
      return { success: false, message: "❌ JSON mein ItemRow nahi mila." };
    }
    if (!Array.isArray(itemRows)) itemRows = [itemRows];

    stage = 'Validation';
    _setProgress(jobId, 32, '✅ Rows validate',
      itemRows.length + ' rows me se valid filter ho rahe hain...');
    var allValidItems = itemRows.filter(function(r) {
      return r && (r.BSODateData || r.BSOItemData);
    });
    if (allValidItems.length === 0) {
      _setProgressDone(jobId, 100, '❌ Valid data nahi', '');
      return { success: false, message: "❌ Koi valid data nahi mila." };
    }

    stage = 'Date range';
    _setProgress(jobId, 36, '📅 Date range',
      allValidItems.length + ' rows ka range nikal raha hu...');
    var allDints = allValidItems.map(function(r){ return dateToInt(r.BSODateData); }).filter(Boolean);
    allDints.sort(function(a,b){ return a-b; });
    var deleteRangeStart = allDints[0];
    var deleteRangeEnd   = allDints[allDints.length - 1];

    stage = 'Voucher filter';
    var validItems = allValidItems;
    if (selectedVoucherTypes && selectedVoucherTypes.length > 0) {
      _setProgress(jobId, 40, '🏷️ Voucher filter',
        selectedVoucherTypes.length + ' types selected');
      var vSet = {};
      selectedVoucherTypes.forEach(function(v){ vSet[v.trim().toLowerCase()] = 1; });
      validItems = allValidItems.filter(function(r){
        return r.BSOVoucherTypeData && vSet[r.BSOVoucherTypeData.toString().trim().toLowerCase()];
      });
      if (validItems.length === 0) {
        _setProgressDone(jobId, 100, '❌ Filter ke baad data nahi', '');
        return { success: false, message: "❌ Selected voucher types mein koi data nahi mila." };
      }
    }

    stage = 'Duplicate detection';
    if (!allowDuplicates) {
      var dupes = _detectDuplicatesAgainstSheet(sheet, validItems, type, jobId);
      if (dupes.length > 0) {
        _setProgressDone(jobId, 100, '⚠️ Duplicate bills mile',
          dupes.length + ' bills already mojud hain alag date ke saath');
        return {
          success: false,
          duplicate: true,
          message: '⚠️ Duplicate detected: ' + dupes.length + ' bill(s) already exist with different dates',
          duplicates: dupes
        };
      }
    }

    var newRows = [];

    if (type === "COMPACT") {
      stage = 'Group bills (COMPACT)';
      var bills = groupByBill(validItems, jobId);

      stage = 'Build rows (COMPACT)';
      _setProgress(jobId, 60, '📦 Rows build',
        bills.length + ' bills me se rows banayi ja rahi hain...');
      var headers = ["Date", "Bill No", "Party Name", "Voucher Type", "Items_JSON", "Narration"];
      bills.forEach(function(b, idx) {
        if (idx % 100 === 0) {
          lastItem = b.party + ' / ' + b.billNo;
          _setProgress(jobId, 60 + Math.round(idx / bills.length * 4),
            '📦 Rows build', 'Bill ' + (idx+1) + '/' + bills.length + ' — ' + lastItem);
        }
        try {
          newRows.push([
            tallyDateToDisplay(b.date),
            b.billNo,
            b.party,
            b.vchType,
            JSON.stringify(b.items),
            b.narr
          ]);
        } catch(rowErr) {
          throw new Error('Bill banane me error: ' + b.party + ' / ' + b.billNo + ' — ' + rowErr.toString());
        }
      });

      stage = 'Write to sheet';
      writeToSheet(sheet, headers, newRows, deleteRangeStart, deleteRangeEnd, jobId);

      _setProgressDone(jobId, 100, '✅ Done!', bills.length + ' bills update ho gayi');
      return {
        success: true,
        message: "✅ Done! [COMPACT]\n" +
                 "🗓 Range: " + intToDisplay(deleteRangeStart) + " → " + intToDisplay(deleteRangeEnd) + "\n" +
                 "🧾 Bills: " + bills.length + " rows"
      };
    }
    else if (type === "DETAILED") {
      stage = 'Build rows (DETAILED)';
      _setProgress(jobId, 50, '📋 Rows build',
        validItems.length + ' rows tayar ho rahi hain...');
      var headers = [
        "Date", "Bill No", "Stock Item", "Godown", "Batch No", "Mfg Date", "Exp Date",
        "Qty In", "Qty Out", "Rate", "Value", "Tax Rate", "Bill Total",
        "Party Name", "Voucher Type", "Narration", "MRP"
      ];

      validItems.forEach(function(item, idx) {
        if (idx % 200 === 0) {
          lastItem = (item.BSOPartyData || '?') + ' / ' + (item.BSOBillNoData || '?') + ' / ' + (item.BSOItemData || '?');
          _setProgress(jobId, 50 + Math.round(idx / validItems.length * 12),
            '📋 Rows build',
            'Row ' + (idx+1) + '/' + validItems.length + ' — ' + lastItem);
        }
        try {
          var rawVal = parseFloat(item.BSOTotalValueData) || 0;
          var absVal = rawVal !== 0 ? Math.abs(rawVal) : "";
          newRows.push([
            tallyDateToDisplay(item.BSODateData    || ""),
            item.BSOBillNoData                     || "",
            item.BSOItemData                       || "",
            item.BSOGodownData                     || "",
            item.BSOBatchNoData                    || "",
            tallyDateToDisplay(item.BSOMfgDateData || ""),
            tallyDateToDisplay(item.BSOExpDateData || ""),
            item.BSOQtyInData                      || "",
            item.BSOQtyOutData                     || "",
            item.BSORateData                       || "",
            absVal,
            item.BSOTaxRateData                    || "",
            item.BSOTaxAmountData                  || "",
            item.BSOPartyData                      || "",
            item.BSOVoucherTypeData                || "",
            item.BSONarrationData                  || "",
            item.BSOMRPData                        || ""
          ]);
        } catch(rowErr) {
          throw new Error('Row banane me error: ' + lastItem + ' — ' + rowErr.toString());
        }
      });

      stage = 'Write to sheet';
      writeToSheet(sheet, headers, newRows, deleteRangeStart, deleteRangeEnd, jobId);

      _setProgressDone(jobId, 100, '✅ Done!', newRows.length + ' rows update ho gayi');
      return {
        success: true,
        message: "✅ Done! [DETAILED]\n" +
                 "🗓 Range: " + intToDisplay(deleteRangeStart) + " → " + intToDisplay(deleteRangeEnd) + "\n" +
                 "📋 Rows: " + newRows.length
      };
    }
    else {
      _setProgressDone(jobId, 100, '❌ Unknown type', type);
      return { success: false, message: "❌ Unknown type: '" + type + "'. App_Settings mein COMPACT ya DETAILED likhein." };
    }

  } catch(e) {
    _setProgressDone(jobId, 100, '❌ Error in: ' + stage,
      (lastItem ? 'Last: ' + lastItem + ' — ' : '') + e.toString());
    return {
      success: false,
      message: "❌ Stage [" + stage + "] me error: " + e.toString() +
               (lastItem ? '\n📍 Last processed: ' + lastItem : '')
    };
  }
}
